    function cleanName(v){ return String(v || '').trim().replace(/\s+/g,' ').slice(0,18); }

    function toast(text,tone='good'){
      const colors = { good:'#4ade80', warn:'#fbbf24', bad:'#f87171' };
      const el = document.createElement('div');

    function escapeHtml(str){ return String(str).replaceAll('&','&amp;').replaceAll('<','&lt;').replaceAll('>','&gt;').replaceAll('"','&quot;').replaceAll("'",'&#039;'); }
    function addLog(text){
      app.state.logs.unshift({ text, time: Date.now() });

    function saveName(){ app.myName = cleanName(els.playerName.value); localStorage.setItem('fahmoh_name', app.myName); return app.myName; }
    function makePlayer(name,isHost=false,id=app.myId){ return { id, name, isHost, score:0, xp:0, level:1, streak:0 }; }

    function ensurePeer(){
      return new Promise((resolve,reject) => {
        if (app.peer && !app.peer.destroyed) return resolve(app.peer);
        els.connText.textContent = 'Connecting';
        app.peer = new Peer();
        app.peer.on('open', id => {
          app.myId = id;
          els.connText.textContent = 'Connected';
          app.peer.on('connection', handleIncomingConnection);
          resolve(app.peer);
        });
        app.peer.on('error', err => {
          console.error(err);
          els.connText.textContent = 'Error';
          toast('Peer connection failed.','bad');
          reject(err);
        });
      });
    }

    async function createRoom(){
      const name = saveName();
      if (!name) return toast('Type your name first.','warn');
      await ensurePeer();
      app.isHost = true;
      app.peers.clear();
      app.roomCode = roomCodeFromId(app.myId);
      app.state = { roomCode:app.roomCode, hostId:app.myId, currentGame:'', currentGameData:null, players:{ [app.myId]: makePlayer(name,true) }, logs:[] };
      addLog(`${name} created the room.`);
      broadcastState();
      render();
      toast(`Room ${app.roomCode} created.`);
    }

    async function joinRoom(){
      const name = saveName();
      const code = String(els.roomCodeInput.value || '').trim();
      if (!name) return toast('Type your name first.','warn');
      if (!code) return toast('Enter a room code.','warn');
      await ensurePeer();
      app.isHost = false;
      const conn = app.peer.connect(code);
      app.hostConn = conn;
      conn.on('open', () => {
        app.roomCode = code.toUpperCase();
        conn.send({ type:'join-request', player:{ id: app.myId, name } });
        render();
      });
      conn.on('data', data => handleClientMessage(data));
      conn.on('close', () => { toast('Disconnected from room.','warn'); leaveRoom(false); });
      conn.on('error', () => toast('Could not join that room.','bad'));
    }

    function handleIncomingConnection(conn){
      if (!app.isHost){ conn.close(); return; }
      conn.on('data', data => handleHostMessage(conn, data));
      conn.on('close', () => {
        const entry = [...app.peers.entries()].find(([,c]) => c === conn);
        if (entry){
          const [id] = entry;
          const name = app.state.players[id]?.name || 'Player';
          app.peers.delete(id);
          delete app.state.players[id];
          addLog(`${name} left the room.`);
          broadcastState();
          render();
        }
      });
    }

    function handleHostMessage(conn, data){
      if (!data || typeof data !== 'object') return;
      if (data.type === 'join-request'){
        const p = data.player;
        app.peers.set(p.id, conn);
        app.state.players[p.id] = makePlayer(p.name,false,p.id);
        addLog(`${p.name} joined the room.`);
        sendState(conn);
        broadcastState();
        render();
      }
      if (data.type === 'action') hostAction(data.action, data.payload, data.playerId);
    }

    function handleClientMessage(data){
      if (!data || typeof data !== 'object') return;
      if (data.type === 'state'){
        app.state = data.state;
        app.roomCode = app.state.roomCode || app.roomCode;
        render();
        syncGameTimer();
      }
    }

    function leaveRoom(showToast=true){
      clearInterval(app.timer);
      app.timer = null;
      if (app.hostConn){ try { app.hostConn.close(); } catch {} }
      for (const conn of app.peers.values()) try { conn.close(); } catch {}
      app.peers.clear();
      app.hostConn = null;
      app.roomCode = '';
      app.isHost = false;
      app.state = { roomCode:'', hostId:'', currentGame:'', currentGameData:null, players:{}, logs:[] };
      render();
      if (showToast) toast('You left the room.','warn');
    }

    function openGame(id){
      if (!app.roomCode) return toast('Create or join a room first.','warn');
      if (app.isHost){
        app.state.currentGame = id;
        app.state.currentGameData = createInitialGameData(id);
        resetScores();
        addLog(`${getGameMeta(id).name} started.`);
        broadcastState();
      } else {
        sendAction('open-game', { id });
      }
      els.hubScreen.classList.remove('active');
      els.gameScreen.classList.add('active');
      render();
      syncGameTimer();
    }

    function resetCurrentGame(){
      if (!app.isHost || !app.state.currentGame) return;
      app.state.currentGameData = createInitialGameData(app.state.currentGame);
      resetScores();
      addLog(`${getGameMeta(app.state.currentGame).name} reset.`);
      broadcastState();
      render();
      syncGameTimer();
    }

    function resetScores(){
      Object.values(app.state.players).forEach(p => { p.score = 0; p.xp = 0; p.level = 1; p.streak = 0; });

      if (id === 'mcdonalds') return { money:0, orders:[randFood(), randFood()], served:0, feed:[] };
      if (id === 'axolotl') return { hunger:70, happiness:70, cleanliness:70, feed:[] };
      if (id === 'rizz') return { prompt: randomRizz(), feed:[] };
      if (id === 'sigma') return { target:100, feed:[] };
      if (id === 'military') return { enemyHP:250, playerHP:Object.fromEntries(Object.keys(app.state.players).map(id => [id,100])), log:['Battle started.'] };
      if (id === 'aura') return { orb:0, feed:[] };
      return {};
    }

    function randomRizz(){ return RIZZ_LINES[Math.floor(Math.random()*RIZZ_LINES.length)]; }
    function randFood(){ return FOODS[Math.floor(Math.random()*FOODS.length)]; }
    function buildWordList(count,difficulty){
      let pool = [...WORDS.easy, ...WORDS.medium, ...WORDS.hard];
      if (difficulty === 'easy') pool = WORDS.easy; else if (difficulty === 'medium') pool = WORDS.medium; else if (difficulty === 'hard') pool = WORDS.hard;
      return [...new Set(pool)].sort(() => Math.random() - 0.5).slice(0,count);
    }

    function sendAction(action,payload={}){
      if (app.isHost) return hostAction(action,payload,app.myId);
      if (app.hostConn && app.hostConn.open) app.hostConn.send({ type:'action', action, payload, playerId: app.myId });
    }

    function hostAction(action,payload,playerId){
      const game = app.state.currentGame;
      const data = app.state.currentGameData;
      const player = app.state.players[playerId];
      if (!player) return;

      if (action === 'open-game' && app.isHost){
        app.state.currentGame = payload.id;
        app.state.currentGameData = createInitialGameData(payload.id);
        resetScores();
        addLog(`${getGameMeta(payload.id).name} started.`);
      }

      if (game === 'spelling'){
        if (action === 'spelling-start'){
          data.round = 1; data.current = data.words[0]; data.status = 'playing'; data.roundEndsAt = Date.now() + 15000; data.results = [];
        }
        if (action === 'spelling-submit' && data.status === 'playing'){
          if (player._spellRound === data.round) return;
          const correct = normalize(payload.answer) === normalize(data.current);
          player._spellRound = data.round;
          let earned = 0;
          if (correct){ earned = 100; player.streak += 1; if (player.streak >= 3) earned += 20; player.score += earned; player.xp += 50; }
          else player.streak = 0;
          player.level = Math.floor(player.xp / 200) + 1;
          data.results.unshift({ name: player.name, answer: payload.answer, correct, earned });
        }
        if (action === 'spelling-next'){
          if (data.round >= data.total){ data.status = 'finished'; data.roundEndsAt = 0; }
          else { data.round += 1; data.current = data.words[data.round - 1]; data.results = []; data.status = 'playing'; data.roundEndsAt = Date.now() + 15000; Object.values(app.state.players).forEach(p => delete p._spellRound); }
        }
      }

      if (game === 'mcdonalds'){
        if (action === 'cook-food'){
          const want = data.orders[0];
          if (payload.food === want){ data.money += 10; player.score += 10; player.xp += 10; data.served += 1; data.feed.unshift(`${player.name} served ${want}.`); data.orders.shift(); data.orders.push(randFood()); }
          else { data.feed.unshift(`${player.name} made ${payload.food} instead of ${want}.`); }
          player.level = Math.floor(player.xp / 200) + 1;
        }
      }

      if (game === 'axolotl'){
        if (action === 'feed-axo'){ data.hunger = Math.min(100, data.hunger + 12); player.score += 8; player.xp += 8; data.feed.unshift(`${player.name} fed the axolotl.`); }
        if (action === 'play-axo'){ data.happiness = Math.min(100, data.happiness + 12); player.score += 8; player.xp += 8; data.feed.unshift(`${player.name} played with the axolotl.`); }
        if (action === 'clean-axo'){ data.cleanliness = Math.min(100, data.cleanliness + 14); player.score += 8; player.xp += 8; data.feed.unshift(`${player.name} cleaned the tank.`); }
        player.level = Math.floor(player.xp / 200) + 1;
      }

      if (game === 'rizz'){
        if (action === 'rizz-pick'){
          const line = payload.score || 0;
          player.score += line; player.xp += Math.max(5, Math.floor(line / 2)); player.level = Math.floor(player.xp / 200) + 1;
          data.feed.unshift(`${player.name} dropped a line for +${line}.`);
          data.prompt = randomRizz();
        }
      }

      if (game === 'sigma'){
        if (action === 'sigma-train'){ player.score += 8; player.xp += 12; data.feed.unshift(`${player.name} trained discipline.`); }
        if (action === 'sigma-focus'){ player.score += 12; player.xp += 16; data.feed.unshift(`${player.name} locked in.`); }
        player.level = Math.floor(player.xp / 200) + 1;
      }

      if (game === 'military'){
        if (!data.playerHP[playerId]) data.playerHP[playerId] = 100;
        if (action === 'mil-attack'){ data.enemyHP = Math.max(0, data.enemyHP - 14); player.score += 14; player.xp += 10; data.log.unshift(`${player.name} attacked for 14.`); }
        if (action === 'mil-heavy'){ data.enemyHP = Math.max(0, data.enemyHP - 24); player.score += 24; player.xp += 15; data.log.unshift(`${player.name} used heavy attack for 24.`); }
        if (action === 'mil-defend'){ data.playerHP[playerId] = Math.min(100, data.playerHP[playerId] + 8); data.log.unshift(`${player.name} defended and recovered 8 HP.`); }
        if (data.enemyHP > 0 && ['mil-attack','mil-heavy','mil-defend'].includes(action)){
          const dmg = 8;
          Object.keys(data.playerHP).forEach(id => data.playerHP[id] = Math.max(0, data.playerHP[id] - dmg));
          data.log.unshift(`Enemy hit everyone for ${dmg}.`);
        }
        player.level = Math.floor(player.xp / 200) + 1;
      }

      if (game === 'aura'){
        if (action === 'aura-meditate'){ player.score += 8; player.xp += 10; data.orb += 1; data.feed.unshift(`${player.name} meditated.`); }
        if (action === 'aura-flex'){ player.score += 12; player.xp += 12; data.orb += 2; data.feed.unshift(`${player.name} did an aura flex.`); }
        if (action === 'aura-absorb' && data.orb > 0){ const gain = data.orb * 5; player.score += gain; player.xp += gain; data.feed.unshift(`${player.name} absorbed ${data.orb} orbs.`); data.orb = 0; }
        player.level = Math.floor(player.xp / 200) + 1;
      }

      broadcastState();
      render();
      syncGameTimer();
    }

    function broadcastState(){
      if (!app.isHost) return;
      const packet = { type:'state', state: app.state };
      for (const conn of app.peers.values()) if (conn.open) conn.send(packet);
    }
    function sendState(conn){ if (conn && conn.open) conn.send({ type:'state', state: app.state }); }

    function syncGameTimer(){
      clearInterval(app.timer);
      app.timer = null;
      const game = app.state.currentGame;
      const data = app.state.currentGameData;
      if (game !== 'spelling' || !data || data.status !== 'playing' || !data.roundEndsAt){ els.statTimer.textContent = '--'; return; }
      const tick = () => {
        const left = Math.max(0, data.roundEndsAt - Date.now());
        els.statTimer.textContent = String(Math.ceil(left / 1000));
        if (left <= 0){ clearInterval(app.timer); app.timer = null; if (app.isHost) hostAction('spelling-next', {}, app.myId); }
      };
      tick();
      app.timer = setInterval(tick, 200);
    }

    function sayWord(){
      const data = app.state.currentGameData;
      if (!data?.current) return;
      if (!('speechSynthesis' in window)) return toast('Speech is not supported here.','warn');
      speechSynthesis.cancel();
      const u = new SpeechSynthesisUtterance(data.current);
      u.lang = 'en-GB';
      u.rate = .9;
      speechSynthesis.speak(u);
    }

    function startScene(){
      const canvas = els.sceneCanvas;
      const renderer = new THREE.WebGLRenderer({ canvas, antialias:true, alpha:true });
      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.setSize(canvas.clientWidth, canvas.clientHeight, false);
      const scene = new THREE.Scene();
      const camera = new THREE.PerspectiveCamera(50, canvas.clientWidth / canvas.clientHeight, .1, 100);
      camera.position.set(0, 1.2, 4.6);
      const ambient = new THREE.AmbientLight(0xffffff, 1.4); scene.add(ambient);
      const dir = new THREE.DirectionalLight(0xffffff, 1.2); dir.position.set(3,4,3); scene.add(dir);
      const group = new THREE.Group(); scene.add(group);
      const floor = new THREE.Mesh(new THREE.CircleGeometry(2.2, 32), new THREE.MeshStandardMaterial({ color:0x152033, metalness:.1, roughness:.8 }));
      floor.rotation.x = -Math.PI/2; floor.position.y = -.9; scene.add(floor);
      app.scene = { renderer, scene, camera, group };
      function loop(){
        requestAnimationFrame(loop);
        group.rotation.y += .01;
        renderer.render(scene, camera);
      }
      loop();
      updateScene('hub');
      window.addEventListener('resize', () => {
        renderer.setSize(canvas.clientWidth, canvas.clientHeight, false);
        camera.aspect = canvas.clientWidth / canvas.clientHeight;
        camera.updateProjectionMatrix();
      });
    }

    function updateScene(mode){
      if (!app.scene) return;
      const { group } = app.scene;
      while (group.children.length) group.remove(group.children[0]);
      const mat1 = new THREE.MeshStandardMaterial({ color:0x22d3ee, metalness:.25, roughness:.45 });
      const mat2 = new THREE.MeshStandardMaterial({ color:0x8b5cf6, metalness:.25, roughness:.45 });
      const mat3 = new THREE.MeshStandardMaterial({ color:0xec4899, metalness:.25, roughness:.45 });
      if (mode === 'military'){
        const body = new THREE.Mesh(new THREE.BoxGeometry(1,1.2,1), mat2); group.add(body);
        const blade = new THREE.Mesh(new THREE.BoxGeometry(.15,1.4,.15), mat1); blade.position.set(1,0,0); group.add(blade);
      } else if (mode === 'aura'){
        for (let i=0;i<6;i++){ const orb = new THREE.Mesh(new THREE.SphereGeometry(.28,24,24), i%2?mat1:mat3); orb.position.set(Math.cos(i)*1.2, Math.sin(i)*.5, Math.sin(i)*1.2); group.add(orb); }
      } else if (mode === 'axolotl'){
        const body = new THREE.Mesh(new THREE.BoxGeometry(1.5,.55,.7), mat1); group.add(body);
        const tail = new THREE.Mesh(new THREE.BoxGeometry(.45,.4,.12), mat3); tail.position.set(-1.05,0,0); group.add(tail);
      } else if (mode === 'mcdonalds'){
        const bun1 = new THREE.Mesh(new THREE.CylinderGeometry(.85,.85,.28,32), new THREE.MeshStandardMaterial({ color:0xd97706 })); bun1.position.y=.45; group.add(bun1);
        const patty = new THREE.Mesh(new THREE.CylinderGeometry(.78,.78,.18,32), new THREE.MeshStandardMaterial({ color:0x5b3716 })); patty.position.y=.08; group.add(patty);
        const bun2 = new THREE.Mesh(new THREE.CylinderGeometry(.85,.85,.28,32), new THREE.MeshStandardMaterial({ color:0xf59e0b })); bun2.position.y=-.25; group.add(bun2);
      } else if (mode === 'spelling'){
        const box = new THREE.Mesh(new THREE.BoxGeometry(1.6,1,1), mat1); group.add(box);
      } else if (mode === 'rizz'){
        const heart = new THREE.Mesh(new THREE.SphereGeometry(.75,24,24), mat3); heart.scale.set(1,.85,.6); group.add(heart);
      } else if (mode === 'sigma'){
        const head = new THREE.Mesh(new THREE.BoxGeometry(1.15,1.15,1.15), mat2); group.add(head);
      } else {
        const cube = new THREE.Mesh(new THREE.BoxGeometry(1.2,1.2,1.2), mat1); group.add(cube);
      }
    }
